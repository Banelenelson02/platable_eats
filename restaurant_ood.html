Employee Hierarchy
abstract class Employee { protected String employeeId, name, role; public Employee(String id, String name, String role) { ... } } class Waiter extends Employee { private RestaurantSystem system; public Order takeOrder(String tableId) { return system.createOrder(tableId, employeeId); } public void submitToKitchen(String orderId) { system.updateOrderStatus(orderId, OrderStatus.IN_KITCHEN); } public void serveMealToTable(String orderId) { system.updateOrderStatus(orderId, OrderStatus.SERVED); } }
Order & OrderItem
class Order { private List<OrderItem> items = new ArrayList<>(); private OrderStatus status = OrderStatus.PENDING; public void addItem(MenuItem item, int qty, String instr) { // Merge if item already exists in order for (OrderItem oi : items) { if (oi.getMenuItem().getId().equals(item.getId())) { oi.setQuantity(oi.getQuantity() + qty); return; } } items.add(new OrderItem(item, qty, instr)); } public double getTotal() { return items.stream().mapToDouble(OrderItem::getSubtotal).sum(); } }
Restaurant System Core
class RestaurantSystem { private Map<String, Table> tables = new LinkedHashMap<>(); private Map<String, MenuItem> menu = new LinkedHashMap<>(); private Map<String, Order> orders = new LinkedHashMap<>(); public Order createOrder(String tableId, String waiterId) { String orderId = "ORD" + String.format("%03d", orderCounter++); Order order = new Order(orderId, tableId, waiterId); orders.put(orderId, order); tables.get(tableId).setStatus(TableStatus.OCCUPIED); return order; } public void updateOrderStatus(String orderId, OrderStatus status) { Order o = orders.get(orderId); if (o != null) { o.setStatus(status); if (status == OrderStatus.SERVED) tables.get(o.getTableId()).setStatus(TableStatus.AVAILABLE); } } }
